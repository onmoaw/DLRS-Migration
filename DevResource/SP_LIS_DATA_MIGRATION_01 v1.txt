MIGRATE_LANDINFO_TO_LINFO00:
-- [dbo].[SP_LIS_DATA_MIGRATION_01] @ComCod = '4103', @ProcID = 'MTO_LINFO00', @Desc01='joarsahara_u';

begin

	create table #templf00 (comcod nchar(4), rsnum nchar(6), unitcod nchar(9), khtnum nvarchar(20), ekhtnum nvarchar(20), pkhtnum nvarchar(20), shetnum nvarchar(5),ltcode nchar(2), kremark nvarchar(1000), khtnote nvarchar(100), khtdat1 xml, khtdat2 xml, khtdat3 xml, khtdat4 xml, khtdat5 xml, khtdat6 xml, khtstatus nvarchar(250), 	usrid nchar(12), sealid nchar(3), sealdesc nvarchar(250), usrses nvarchar(10), termid nvarchar(50), rowtime datetime);

  
	declare @DynamicQuerylf00 nvarchar(max);
	set @DynamicQuerylf00 = N'insert into #templf00 select @comcod as comcod, @desc02 as rsnum, ltrim(rtrim(a.unitcod)) as unitcod, ltrim(rtrim(a.khtnum)) as khtnum, ltrim(rtrim(a.KhtNum)) as ekhtnum,N'''' as pkhtnum, N''000'' AS shetnum, ltrim(rtrim(a.LndTyp)) as ltcode,ltrim(rtrim(a.KRemark)) as kremark,N'''' as khtnote,NULL AS khtdat1,NULL as khtdat2,NULL as khtdat3,NULL as khtdat4,NULL as khtdat5,NULL as khtdat6,N'''' as khtstatus,ltrim(rtrim(a.UsrId)) as usrid,N''000'' AS sealid,N'''' as sealdesc,N'''' as usrses,N'''' as termid,ltrim(rtrim(a.Entdate)) as ROWTIME from '+quotename(@desc01)+'.dbo.tbllinfo01 a';	
	 exec sp_executesql @DynamicQuerylf00,  N'@ComCod nchar(4), @Desc02 nvarchar(50)',  @ComCod, @Desc02;   
     

	with temp001 as ( select  a.comcod, a.rsnum, a.unitcod, a.khtnum, a.ekhtnum, a.pkhtnum, a.shetnum, a.ltcode, ISNULL(a.kremark, '') as kremark,  ISNULL(a.khtnote, '') as khtnote, ISNULL(a.khtstatus, '') as khtstatus, a.usrid,a.sealid, a.sealdesc, a.usrses, a.termid, a.rowtime, a.khtdat1, a.khtdat2, a.khtdat3, a.khtdat4, a.khtdat5, a.khtdat6,
        ROW_NUMBER() over ( partition by a.comcod, a.rsnum, a.unitcod, a.khtnum  order by a.rowtime desc) as newrownum from #templf00 a)
    insert into DLRSLSINF_4103PRSDB.dbo_lis.linfo00 (comcod, rsnum, unitcod, khtnum, ekhtnum, pkhtnum, shetnum, ltcode, kremark, khtnote,  khtdat1, khtdat2, khtdat3, khtdat4, khtdat5, khtdat6, khtstatus, usrid, sealid,  sealdesc, usrses, termid, rowtime)

    drop table #templf00;	
return;


end; -- MIGRATE_LANDINFO_TO_LINFO00:

MIGRATE_LANDINFO_TO_LINFO01:
-- [dbo].[SP_LIS_DATA_MIGRATION_01] @ComCod = '4103', @ProcID = 'MTO_LINFO01', @Desc01='joarsahara_u', @Desc02='201901';
begin

	create table #templf01 (comcod nchar(4), rsnum nchar(6), unitcod nchar(9), khtnum nvarchar(20), pltnum nvarchar(15), slnum int, lucode nchar(5), tpltarea decimal(18,6), 
	khtshare decimal(18,6), kpltarea decimal(18,6), kpltrmrk nvarchar(1500), skhtnum nvarchar(250), spltnum nvarchar(500), cpltnum nvarchar(250), rowtime datetime);

	declare @DynamicQuerylf05 nvarchar(max);
 
	 SET @DynamicQuerylf05 = N'with cte as (select @comcod as comcod, @desc02 as rsnum, LTRIM(RTRIM(a.unitcod)) as unitcod, LTRIM(RTRIM(a.khtnum)) as khtnum, 
	 LTRIM(RTRIM(a.pltnum)) as pltnum, 1 as slnum, N'''' as lucode, 0.0 as tpltarea, 0.0 as khtshare, 0.0 as kpltarea, N'''' as kpltrmrk, N'''' as skhtnum, 
     LTRIM(RTRIM(a.HSPltNum)) as spltnum, N'''' as cpltnum, a.entdt as rowtime FROM '+quotename(@desc01)+'.dbo.tblHalsabek a)
	 INSERT INTO #templf01 (comcod, rsnum, unitcod, khtnum, pltnum, slnum, lucode, tpltarea, khtshare, kpltarea, kpltrmrk, skhtnum, spltnum, cpltnum, rowtime) 
	 SELECT comcod, rsnum, unitcod, khtnum, pltnum, MAX(slnum), MAX(lucode) as lucode, MAX(tpltarea) as tpltarea, MAX(khtshare) as khtshare, MAX(kpltarea) as kpltarea,  
	 MAX(kpltrmrk) as kpltrmrk, MAX(skhtnum) as skhtnum, STRING_AGG(spltnum, '','') as spltnum, MAX(cpltnum) as cpltnum, MAX(rowtime) as rowtime FROM CTE GROUP BY comcod, rsnum, unitcod, khtnum, pltnum;';

	EXEC sp_executesql @DynamicQuerylf05, N'@ComCod nchar(4), @Desc02 nvarchar(50)',  @ComCod, @Desc02;

	select*from #templf01;


	DECLARE @DynamicQuerylf06 NVARCHAR(MAX); 

	SET @DynamicQuerylf06 = N'update a set a.tpltarea = b.TPltArea, a.KHTSHARE=b.KhtShare, a.KPLTAREA=b.KPltArea, a.KPLTRMRK=b.KRemark
	from #templf01 a inner join ' + QUOTENAME(@desc01) + '.dbo.tblLinfo01 b on a.unitcod=b.unitcod and a.khtnum=b.khtnum and a.pltnum=b.pltnum';

	EXEC sp_executesql @DynamicQuerylf06, N'@ComCod nchar(4), @Desc02 nvarchar(50)', @ComCod, @Desc02;

	SELECT * FROM #templf01;


RETURN;
END;

-- [dbo].[SP_LIS_DATA_MIGRATION_01] @ComCod = '4103', @ProcID = 'MTO_PLOTINF01', @Desc01='joarsahara_u', @Desc02='201901';

begin

	create table #temppltinf01 (comcod nchar(4), rsnum nchar(6), unitcod nchar(9), shetnum nvarchar(5), pltnum nvarchar(15), tpltarea nvarchar(30), lucode nchar (5));

	set @DynamicQueryplt01 = N'insert into #temppltinf01 select distinct @ComCod as comcod, @Desc02 as rsnum, ltrim(rtrim(a.unitcod)) as unitcod, N''00'' as shetnum,
